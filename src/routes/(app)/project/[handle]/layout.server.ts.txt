// src/routes/project/[handle]/+layout.server.ts
import { redirect } from "@sveltejs/kit";
import type { LayoutServerLoad } from "./$types";
import db from "@db";
import { project, room } from "@db/schema";
import { and, eq } from "drizzle-orm";

export const load: LayoutServerLoad = async ({ locals, params, url }) => {
  console.log("Layout Server - Handle:", params.handle);
  console.log("Layout Server - Handles:", params);
  console.log("Current User:", locals.user?.id);

  if (!locals.user) {
    console.log("No user, redirecting to login");
    throw redirect(302, "/login");
  }

  const urlPathname = url.pathname; // "/project/the-big-move"
  const pathSegments = urlPathname.split("/");
  const projectHandle = pathSegments[pathSegments.length - 1]; // "the-big-move"
  console.log("LAYOUT SERVER HANDLE FOR PROJECT", projectHandle);

  // Fetch the project with the rooms based on the project handle
  let PROJECT = await db.query.project.findFirst({
    where: and(
      eq(project.handle, projectHandle), // Ensure this is the project handle
      eq(project.userId, locals.user.id),
    ),
    with: {
      rooms: {
        with: {
          boxes: {
            with: {
              items: true,
            },
          },
        },
      },
    },
  });

  console.log("Fetched Project:", PROJECT);

  // Verify project exists and is owned by the current user
  if (!PROJECT) {
    console.log("No project found, redirecting to dashboard");
    throw redirect(302, "/dashboard");
  }

  console.log("Project User ID:", PROJECT.userId);
  console.log("Current User ID:", locals.user.id);

  // Ensure strict ownership check
  if (PROJECT.userId !== locals.user.id) {
    console.log("User does not have access to this project");
    throw redirect(302, "/dashboard");
  }

  return {
    project: PROJECT,
  };
};
